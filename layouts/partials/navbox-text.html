{{- /* Compare final Hugo URLs, preserving captions, math, slug and base paths. */ -}} {{- $html := .page.RenderString (dict "display" "inline") .text -}} {{- $current := slice .page.RelPermalink .page.Permalink -}} {{- range findRE "]*>[\\s\\S]*?" $html -}} {{- $anchor := . -}} {{- range findRE "href=\"[^\"]*\"" $anchor -}} {{- $href := . | replaceRE "^href=\"|\"$" "" | htmlUnescape | replaceRE "\\?from=[^&#]*$" "" -}} {{- if in $current $href -}} {{- $text := $anchor | replaceRE "^]*>([\\s\\S]*?)$" "$1" -}} {{- $html = replace $html $anchor (printf "%s" $text) -}} {{- end -}} {{- end -}} {{- end -}} {{- return ($html | safeHTML) -}}
